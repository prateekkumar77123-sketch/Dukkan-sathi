  <script>
    const WEB_URL = 'http://localhost:5173';
    const MCP_URL = 'http://localhost:8001/mcp';

    // ---------- UI helpers ----------
    function log(msg, type = 'info') {
      const el = document.getElementById('console-log');
      const line = document.createElement('div');
      line.className = `log-line log-${type}`;
      const ts = new Date().toLocaleTimeString();
      line.textContent = `[${ts}] ${msg}`;
      el.prepend(line);
    }

    function setStatus(idPrefix, ok, message) {
      const dot = document.getElementById(`${idPrefix}-dot`) || document.getElementById(`dot-${idPrefix}`);
      const text = document.getElementById(`text-${idPrefix}`);
      if (!dot || !text) return;
      if (ok) {
        dot.className = 'dot ok';
        text.textContent = message || 'Online';
        text.style.color = 'var(--ok)';
      } else {
        dot.className = 'dot err';
        text.textContent = message || 'Offline / unreachable';
        text.style.color = 'var(--err)';
      }
    }

    function openUrl(url) {
      window.open(url, '_blank');
    }

    function toggleIframe(id) {
      const iframe = document.getElementById(id);
      const isHidden = iframe.classList.contains('hide');
      if (isHidden) {
        iframe.src = WEB_URL;
        iframe.classList.remove('hide');
        log('Opened website preview iframe.');
      } else {
        iframe.classList.add('hide');
        iframe.src = 'about:blank';
        log('Closed website preview iframe.');
      }
    }

    // ---------- Endpoint probes ----------
    async function probe(url, idPrefix) {
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), 4000);
      try {
        const res = await fetch(url, {
          method: 'GET',
          signal: controller.signal,
          headers: { 'Accept': 'text/plain, application/json' },
        });
        clearTimeout(timeout);
        setStatus(idPrefix, res.ok, `HTTP ${res.status}`);
        log(`${url} responded ${res.status}`, 'ok');
        return true;
      } catch (e) {
        clearTimeout(timeout);
        setStatus(idPrefix, false, e.name === 'AbortError' ? 'Timeout' : 'Unreachable');
        log(`${url} could not be reached: ${e.message}`, 'err');
        return false;
      }
    }

    async function probeAll() {
      await probe(WEB_URL, 'web');
      // MCP usually speaks JSON-RPC over POST; a GET probe tells us if the port is alive
      await probe(MCP_URL, 'mcp');
    }

    // ---------- MCP JSON-RPC helpers ----------
    let mcpId = 0;

    async function mcpRequest(method, params = {}) {
      mcpId++;
      const payload = {
        jsonrpc: '2.0',
        id: mcpId,
        method,
        params,
      };

      const output = document.getElementById('mcp-output');
      output.textContent = `Sending ${method}…`;

      try {
        const res = await fetch(MCP_URL, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Accept': 'application/json, text/event-stream',
          },
          body: JSON.stringify(payload),
        });

        const text = await res.text();
        let parsed;
        try {
          parsed = JSON.parse(text);
        } catch {
          parsed = { raw: text };
        }

        output.textContent = JSON.stringify(parsed, null, 2);
        log(`MCP ${method} → ${res.status}`, 'ok');
        return parsed;
      } catch (e) {
        output.textContent = `Error: ${e.message}`;
        log(`MCP ${method} failed: ${e.message}`, 'err');
        return { error: e.message };
      }
    }

    async function mcpInitialize() {
      setStatus('mcp', false, 'Initializing…');
      const result = await mcpRequest('initialize', {
        protocolVersion: '2024-11-05',
        capabilities: {},
        clientInfo: { name: 'html-dashboard', version: '1.0.0' },
      });
      if (result && result.result) {
        setStatus('mcp', true, 'Initialized');
        log('MCP initialize succeeded.', 'ok');
      } else {
        setStatus('mcp', false, 'Init failed');
      }
    }

    async function mcpListTools() {
      await mcpRequest('tools/list');
    }

    // ---------- Boot ----------
    probeAll();
    setInterval(probeAll, 30000); // re-check every 30s
  </script>
